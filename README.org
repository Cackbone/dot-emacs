#+TITLE: EMACS CONFIG 3.0.0
#+LANGUAGE: en

- Emacs version: *31* (needs 30+: built-in which-key, use-package with ~:vc~)
- Mostly used as a server (~emacs --daemon~), from two fish functions:
  - ~ne [files]~: a plain, quick window;
  - ~nee [file or folder]~: the IDE layout, a large window with the file tree (treemacs) on
    the left and Claude Code on the right, reusing the project's running Claude session.
  Terminal frames: ~emacsclient -t~.
- Keys: ~F8~ file tree, ~F9~ Claude Code, ~C-c C-'~ Claude Code's menu.
- Theme: [[https://github.com/Cackbone/camron-theme][camron]], dark navy with neon magenta and cyan.

The tree's footer: where the project stands in git (project, branch and its ↑↓ to the
remote, modified ● and untracked ✚ files, ✓ when clean), refreshed every 5 s.
#+BEGIN_SRC emacs-lisp
  (defface my/tree-footer-name   '((t :foreground "#e040fb" :weight bold)) "Tree footer: project.")
  (defface my/tree-footer-branch '((t :foreground "#76e6f2")) "Tree footer: branch.")
  (defface my/tree-footer-dirty  '((t :foreground "#ffd166")) "Tree footer: modified files.")
  (defface my/tree-footer-new    '((t :foreground "#76e6f2")) "Tree footer: untracked files.")
  (defface my/tree-footer-clean  '((t :foreground "#69ff47")) "Tree footer: nothing to commit.")
  (defface my/tree-footer-dim    '((t :foreground "#8b8bc7")) "Tree footer: ahead / behind.")

  (defvar-local my/tree-footer nil "(ROOT BRANCH AHEAD BEHIND MODIFIED UNTRACKED), or (ROOT).")

  (defun my/tree-footer--root ()
    (when-let* ((ws (treemacs-current-workspace))
                (project (car (treemacs-workspace->projects ws))))
      (treemacs-project->path project)))

  (defun my/tree-footer-update ()
    "Refresh the git summary of every visible tree."
    (dolist (b (buffer-list))
      (when (and (eq (buffer-local-value 'major-mode b) 'treemacs-mode) (get-buffer-window b t))
        (with-current-buffer b
          (when-let* ((root (my/tree-footer--root)))
            (let* ((default-directory (file-name-as-directory root))
                   (lines (ignore-errors (process-lines "git" "status" "--porcelain=v1" "-b"))))
              (setq my/tree-footer
                    (if (null lines)
                        (list root)
                      (let ((head (car lines)) (files (cdr lines)))
                        (list root
                              (and (string-match "^## \\(?:No commits yet on \\)?\\([^. ]+\\)" head)
                                   (match-string 1 head))
                              (and (string-match "ahead \\([0-9]+\\)" head) (string-to-number (match-string 1 head)))
                              (and (string-match "behind \\([0-9]+\\)" head) (string-to-number (match-string 1 head)))
                              (seq-count (lambda (l) (not (string-prefix-p "??" l))) files)
                              (seq-count (lambda (l) (string-prefix-p "??" l)) files)))))
              (force-mode-line-update)))))))

  (defun my/tree-footer-line ()
    "The tree's mode line; the branch name shortens with … so the counts always show."
    (pcase-let ((`(,root ,branch ,ahead ,behind ,modified ,untracked) my/tree-footer)
                (esc (lambda (s) (string-replace "%" "%%" s))))
      (if (null root)
          ""
        (let* ((name (file-name-nondirectory (directory-file-name root)))
               (sync (concat (if ahead (format " ↑%d" ahead) "") (if behind (format " ↓%d" behind) "")))
               (state (cond ((null branch) "")
                            ((and (eq modified 0) (eq untracked 0)) (propertize "✓" 'face 'my/tree-footer-clean))
                            (t (string-trim-right
                                (concat (if (> modified 0) (propertize (format "●%d " modified) 'face 'my/tree-footer-dirty) "")
                                        (if (> untracked 0) (propertize (format "✚%d" untracked) 'face 'my/tree-footer-new) ""))))))
               ;; room left for the branch: window width − name, separators, icon, ↑↓, counts
               (room (- (window-total-width) 1 (length name) 2 2 (length sync) 2 (length state) 1))
               (br (and branch (if (> (length branch) (max 3 room))
                                   (concat (substring branch 0 (max 2 (1- room))) "…")
                                 branch))))
          (concat " " (propertize (funcall esc name) 'face 'my/tree-footer-name)
                  (when br
                    (concat "  " (propertize (concat "\ue0a0 " (funcall esc br)) 'face 'my/tree-footer-branch)
                            (propertize sync 'face 'my/tree-footer-dim)
                            "  " state)))))))

  (add-hook 'treemacs-mode-hook
            (lambda ()
              (setq-local mode-line-format '((:eval (my/tree-footer-line))))
              (face-remap-add-relative 'default :height 0.85)       ; smaller text, like Claude's
              (run-at-time 0.5 nil #'my/tree-footer-update))
            t)
  (run-with-timer 3 5 #'my/tree-footer-update)
#+END_SRC

** Claude Code
Claude Code in an Emacs buffer that knows about Emacs: it sees the open files and the
selection, shows its edits as diffs (ediff), and can use LSP / flycheck information.
~C-c C-'~ opens its menu (new session, continue, send the region…); ~F9~ shows / hides it. Runs in eat, a terminal
written in Emacs Lisp (nothing to compile).
#+BEGIN_SRC emacs-lisp
  ;; a Claude started here is a normal session even when this Emacs was launched from inside
  ;; one (it inherits CLAUDE_CODE_CHILD_SESSION…: "transcript saving is off", no --resume)
  (setq process-environment
        (seq-remove (lambda (v) (string-match-p "\\`\\(CLAUDE\\|CLAUDECODE\\)" v))
                    process-environment))
  ;; the full-screen renderer, as in a terminal: prompt and status pinned to the window's bottom
  ;; (inside Emacs it otherwise falls back to the inline one, the prompt following the output)
  (setenv "CLAUDE_CODE_NO_FLICKER" "1")

  (use-package eat)

  (use-package claude-code-ide
    :vc (:url "https://github.com/manzaltu/claude-code-ide.el" :rev :newest)
    :bind (("C-c C-'" . claude-code-ide-menu)
           ([f9] . my/claude-code-ide-toggle))
    :config
    (setq claude-code-ide-terminal-backend 'eat)
    ;; its window: 1.5 × treemacs's width (the tree's window when it's open, else its width
    ;; setting), worked out each time it opens, and checked once shown
    (defun my/claude-code-ide-width (&rest _)
      (let ((tree (and (fboundp 'treemacs-get-local-window) (treemacs-get-local-window))))
        (round (* 1.5 (if (window-live-p tree)
                          (window-total-width tree)
                        (if (boundp 'treemacs-width) treemacs-width 32))))))
    (advice-add 'claude-code-ide--display-buffer-in-side-window :before
                (lambda (&rest _)
                  (setq claude-code-ide-window-width (my/claude-code-ide-width))))
    (advice-add 'claude-code-ide--display-buffer-in-side-window :filter-return
                (lambda (win)
                  (when (window-live-p win)
                    (let ((delta (- (my/claude-code-ide-width) (window-total-width win))))
                      (unless (zerop delta)
                        (ignore-errors (window-resize win delta t t))
                        ;; tell the terminal (and so Claude) its new size: the package synced
                        ;; it before this resize, and lines would wrap a few columns too late
                        (my/claude-sync-size (window-buffer win)))))
                  win))
    (claude-code-ide-emacs-tools-setup))

  ;; F9 (next to treemacs' F8): show / hide the project's Claude, starting one if it has none
  (defun my/claude-code-ide-toggle ()
    "Toggle this project's Claude Code window; start a session when there's none."
    (interactive)
    (require 'claude-code-ide)
    (if (claude-code-ide-mcp--sessions-for-project (claude-code-ide--get-working-directory))
        (claude-code-ide-toggle)
      (claude-code-ide)))
#+END_SRC

*** Usage bars
The context and 5-hour session bars sit in the Claude window's mode line, where they never
wrap. Each Claude buffer reads them every 3 s from a file named after its Claude process,
~$XDG_RUNTIME_DIR/claude-emacs/<pid>.json~: ~{"ctx": 23, "q5": 5, "reset": "16:50"}~ (context
%, 5-hour session %, its reset time). Claude Code's status line command writes it: it gets
those numbers from Claude Code (~.context_window.used_percentage~,
~.rate_limits.five_hour~), and can tell it runs here from ~$INSIDE_EMACS~. Without that
file, the line says it's waiting.
#+BEGIN_SRC emacs-lisp
  (defface my/claude-usage-label '((t :foreground "#8b8bc7")) "Usage bar labels.")
  (defface my/claude-usage-empty '((t :foreground "#3b2a78")) "Usage bars, empty part.")
  (defface my/claude-usage-high  '((t :foreground "#ff4f79")) "Usage bars past 80%.")

  (defvar-local my/claude-usage nil "This Claude's last usage: ctx, q5, reset.")
  (defvar-local my/small-text nil "The face remapping that shrinks this buffer's text.")

  (defun my/claude-sync-size (buffer)
    "Give BUFFER's terminal, and the Claude running in it, the size its window really fits.
A font change doesn't resize the window, so nothing would tell them otherwise."
    (when-let* ((win (get-buffer-window buffer t))
                (proc (get-buffer-process buffer)))
      (claude-code-ide--sync-terminal-dimensions buffer win)
      (with-current-buffer buffer
        (when (bound-and-true-p eat-terminal)
          (let ((size (eat-term-size eat-terminal)))
            (set-process-window-size proc (cdr size) (car size)))))))

  (defun my/claude-usage--bar (label pct cells &optional note)
    "LABEL, a bar of CELLS for PCT (cyan → magenta, red past 80%), the percentage, NOTE."
    (let* ((pct (max 0 (min 100 pct)))
           (n (round (* cells (/ pct 100.0))))
           (bar (mapconcat
                 (lambda (i)
                   (propertize "━" 'face
                               (cond ((>= i n) 'my/claude-usage-empty)
                                     ((>= pct 80) 'my/claude-usage-high)
                                     (t (let ((x (/ (float i) (max 1 (1- cells)))))
                                          (list :foreground
                                                (format "#%02x%02x%02x" (round (* x #xe0))
                                                        (round (+ #xbc (* x (- #x40 #xbc))))
                                                        (round (+ #xd4 (* x (- #xfb #xd4)))))))))))
                 (number-sequence 0 (1- cells)) "")))
      (concat (propertize label 'face 'my/claude-usage-label) " " bar " "
              ;; "%%%%": the mode line reads "%" as a code of its own, so it's doubled once more
              (propertize (format "%d%%%%" pct) 'face (if (>= pct 80) 'my/claude-usage-high 'my/claude-usage-label))
              (if note (propertize (concat " ↻ " note) 'face 'my/claude-usage-label) ""))))

  (defun my/claude-usage-line ()
    "Mode line of a Claude window: its context and 5-hour session bars."
    (let ((u my/claude-usage))
      (if (null u)
          (propertize "  waiting for Claude's first status…" 'face 'my/claude-usage-label)
        (concat "  " (my/claude-usage--bar "ctx" (or (alist-get 'ctx u) 0) 10)
                (when-let* ((q5 (alist-get 'q5 u)))
                  (concat "   " (my/claude-usage--bar "5h" q5 8
                                                     (let ((r (alist-get 'reset u)))
                                                       (and (stringp r) (> (length r) 0) r)))))))))

  (defun my/claude-usage-refresh ()
    "Read each Claude buffer's usage file; redraw its mode line when it changed."
    (dolist (b (buffer-list))
      (when (buffer-local-value 'my/claude-usage-mode b)
        (when-let* ((proc (get-buffer-process b))
                    (pid (process-id proc))
                    (f (expand-file-name (format "claude-emacs/%d.json" pid)
                                         (or (getenv "XDG_RUNTIME_DIR") "/tmp")))
                    ((file-readable-p f))
                    (data (ignore-errors
                            (json-parse-string (with-temp-buffer (insert-file-contents f) (buffer-string))
                                               :object-type 'alist :null-object nil))))
          (unless (equal data (buffer-local-value 'my/claude-usage b))
            (with-current-buffer b
              (setq my/claude-usage data)
              (force-mode-line-update)))))))

  (define-minor-mode my/claude-usage-mode
    "Show this Claude's context and session bars as the window's mode line."
    :lighter nil
    (if my/claude-usage-mode
        (progn
          (setq-local mode-line-format '((:eval (my/claude-usage-line))))
          ;; the wheel goes to Claude (its full-screen view scrolls the conversation and keeps
          ;; the prompt in place), not to Emacs's smooth scrolling of the whole buffer
          (setq-local pixel-scroll-precision-mode nil)
          ;; never narrower than its default width (dragging the border stops there)
          (setq-local window-min-width (my/claude-code-ide-width))
          ;; smaller text than the files (85%): more fits in the side panel
          (setq-local my/small-text (face-remap-add-relative 'default :height 0.85))
          (let ((buf (current-buffer))) (run-at-time 0.3 nil #'my/claude-sync-size buf)))
      (kill-local-variable 'mode-line-format)
      (kill-local-variable 'pixel-scroll-precision-mode)
      (kill-local-variable 'window-min-width)
      (when (bound-and-true-p my/small-text) (face-remap-remove-relative my/small-text))))

  (run-with-timer 2 3 #'my/claude-usage-refresh)
  ;; every Claude window gets it when it's shown
  (with-eval-after-load 'claude-code-ide
    (advice-add 'claude-code-ide--display-buffer-in-side-window :filter-return
                (lambda (win)
                  (when (window-live-p win)
                    (with-current-buffer (window-buffer win)
                      (unless my/claude-usage-mode (my/claude-usage-mode 1))))
                  win)))
#+END_SRC

*** IDE mode
What the fish function ~nee~ opens: a large window with the file (or the folder, in dired),
treemacs with its project on the left, Claude on the right (1.5 × treemacs's width). Claude
is the project's running session if there is one in this Emacs, else a fresh one.
#+BEGIN_SRC fish
function nee --description 'Emacs in IDE mode: the file or folder, treemacs and Claude Code'
    set -l target (path resolve -- (test (count $argv) -gt 0; and echo $argv[1]; or echo .))
    set target (string replace -a '\\' '\\\\' -- $target | string replace -a '"' '\\"')
    emacsclient -nc -a "" -F '((fullscreen . maximized))' -e "(my/ide \"$target\")" >/dev/null &
end
#+END_SRC

#+BEGIN_SRC emacs-lisp
  (defun my/ide (&optional path)
    "Open PATH (a file, or a folder in dired) with treemacs and Claude Code beside it."
    (let ((path (expand-file-name (or path default-directory))))
      (if (file-directory-p path) (dired path) (find-file path))
      (delete-other-windows)
      (let ((main (selected-window)))
        ;; the tree: this project only
        (save-selected-window (treemacs-add-and-display-current-project-exclusively))
        ;; Claude: the project's running session, else a new one
        (require 'claude-code-ide)
        (if (claude-code-ide-mcp--sessions-for-project (claude-code-ide--get-working-directory))
            (claude-code-ide-switch-to-buffer)
          (claude-code-ide))
        (when (window-live-p main) (select-window main)))))
#+END_SRC
